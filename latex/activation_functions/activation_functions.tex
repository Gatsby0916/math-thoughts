\section{Why we need them}

A neuron computes a pre-activation and then applies an activation function $\phi$:
\begin{equation}
  z = w^\top x + b, \qquad a = \phi(z).
\end{equation}
Without $\phi$, stacking layers gives nothing new:
\begin{equation}
  W_2(W_1x+b_1)+b_2 = (W_2W_1)\,x + (W_2b_1+b_2),
\end{equation}
which is still one linear map. So depth only helps when $\phi$ is nonlinear.

For training, $\phi'$ matters as much as $\phi$. Take a chain with one unit per layer,
$a_l=\phi(z_l)$ and $z_{l+1}=w_{l+1}a_l$. By the chain rule
\begin{equation}
  \frac{\partial a_L}{\partial z_1}
  = \phi'(z_L)\prod_{l=1}^{L-1} w_{l+1}\,\phi'(z_l).
  \label{eq:chain}
\end{equation}
If $|\phi'|$ is small, this product shrinks with $L$ (vanishing gradient).
Below, each function is described by its formula, range, derivative,
and how fast $\phi'$ decays for large $|z|$.

\section{Sigmoid}

\begin{equation}
  \sigma(z)=\frac{1}{1+e^{-z}}, \qquad \sigma(z)\in(0,1), \qquad \sigma(0)=\tfrac12 .
\end{equation}
It is symmetric about the point $(0,\tfrac12)$:
\begin{equation}
  \sigma(-z)=1-\sigma(z).
\end{equation}
The derivative can be written with $\sigma$ itself:
\begin{equation}
  \sigma'(z)=\frac{e^{-z}}{(1+e^{-z})^2}=\sigma(z)\bigl(1-\sigma(z)\bigr),
  \qquad \max_z \sigma'(z)=\sigma'(0)=\tfrac14 .
  \label{eq:sigmoid-derivative}
\end{equation}
For large $|z|$,
\begin{equation}
  1-\sigma(z)=\frac{1}{1+e^{z}}\approx e^{-z}\ (z\to+\infty),
  \qquad \sigma'(z)\approx e^{-|z|}.
\end{equation}

Two problems follow.
First, with $\sigma'\le\tfrac14$, \eqref{eq:chain} gives
\begin{equation}
  \left|\frac{\partial a_L}{\partial z_1}\right|
  \le \left(\tfrac14\right)^{L}\prod_{l=1}^{L-1}|w_{l+1}|,
  \qquad \left(\tfrac14\right)^{10}\approx 9.5\times10^{-7}.
\end{equation}
Second, $\sigma(z)>0$ always. For a unit with inputs $a_j=\sigma(z_j)$ and error signal $\delta$,
\begin{equation}
  \frac{\partial \ell}{\partial w_j}=\delta\,a_j ,
\end{equation}
so all $\partial\ell/\partial w_j$ share the sign of $\delta$ and the weights move in a zig-zag.

\section{Hyperbolic tangent}

\begin{equation}
  \tanh z=\frac{e^{z}-e^{-z}}{e^{z}+e^{-z}}, \qquad \tanh z\in(-1,1), \qquad \tanh 0=0 .
\end{equation}
It is a rescaled sigmoid:
\begin{equation}
  \tanh z = 2\sigma(2z)-1,
  \qquad\text{since}\quad
  2\sigma(2z)-1=\frac{1-e^{-2z}}{1+e^{-2z}} .
  \label{eq:tanh-sigmoid}
\end{equation}
It is odd, so its outputs are centred at $0$. The derivative is
\begin{equation}
  \tanh' z = 1-\tanh^2 z = \frac{4}{(e^{z}+e^{-z})^2},
  \qquad \max_z \tanh' z=\tanh' 0 = 1 ,
\end{equation}
four times the peak of $\sigma'$. For large $|z|$,
\begin{equation}
  1-\tanh z=\frac{2}{e^{2z}+1}\approx 2e^{-2z}\ (z\to+\infty),
  \qquad \tanh' z\approx 4e^{-2|z|},
\end{equation}
so $\tanh$ saturates even faster than $\sigma$. It fixes the sign problem and the small peak, but not saturation.

\section{Arctangent}

\begin{equation}
  \phi(z)=\arctan z, \qquad \arctan z\in\left(-\tfrac{\pi}{2},\tfrac{\pi}{2}\right), \qquad \arctan 0=0 .
\end{equation}
Like $\tanh$ it is odd and zero-centred, with peak slope $1$:
\begin{equation}
  \frac{d}{dz}\arctan z=\frac{1}{1+z^2}, \qquad \max_z=1 \text{ at } z=0 .
\end{equation}
The difference is in the tails. For $z>0$,
\begin{equation}
  \frac{\pi}{2}-\arctan z=\arctan\frac1z\approx\frac1z,
  \qquad \frac{1}{1+z^2}\approx\frac{1}{z^2},
\end{equation}
so the derivative decays like a power of $z$, not exponentially. At $z=5$:
\begin{equation}
  \tanh'(5)\approx1.8\times10^{-4}, \qquad
  \sigma'(5)\approx6.6\times10^{-3}, \qquad
  \arctan'(5)=\tfrac{1}{26}\approx3.8\times10^{-2}.
\end{equation}
Near $0$, $\arctan$ and $\tanh$ are almost the same function:
\begin{equation}
  \tanh z = z-\frac{z^3}{3}+\frac{2z^5}{15}-\cdots, \qquad
  \arctan z = z-\frac{z^3}{3}+\frac{z^5}{5}-\cdots .
  \label{eq:taylor}
\end{equation}
They agree up to $z^3$. All the difference is for large $|z|$.
To get the same range as $\tanh$ one can use $\tfrac{2}{\pi}\arctan z\in(-1,1)$, whose slope at $0$ is $\tfrac{2}{\pi}$.

\section{ReLU}

\begin{equation}
  \operatorname{ReLU}(z)=\max(0,z), \qquad \operatorname{ReLU}(z)\in[0,\infty).
\end{equation}
The derivative is a step:
\begin{equation}
  \operatorname{ReLU}'(z)=
  \begin{cases}
    1, & z>0,\\
    0, & z<0,
  \end{cases}
\end{equation}
and at $z=0$ any value in $[0,1]$ is a subgradient; libraries use $0$.

On active units ($z_l>0$) every factor $\phi'(z_l)$ in \eqref{eq:chain} equals $1$:
\begin{equation}
  \frac{\partial a_L}{\partial z_1}=\prod_{l=1}^{L-1} w_{l+1}
  \qquad\text{if all } z_l>0,
\end{equation}
so the activation itself causes no vanishing. Other properties:
\begin{equation}
  \operatorname{ReLU}(cz)=c\,\operatorname{ReLU}(z)\quad (c\ge0),
  \qquad
  \operatorname{ReLU}(z)=0 \ \text{for } z<0 .
\end{equation}
The first is positive homogeneity. The second gives sparse outputs and costs one comparison,
but a unit with $z<0$ for every input gets zero gradient forever (a ``dead'' unit).
Leaky ReLU keeps a small slope $\alpha$ (often $0.01$) on the negative side:
\begin{equation}
  \operatorname{LReLU}_\alpha(z)=\max(\alpha z, z), \qquad
  \operatorname{LReLU}_\alpha'(z)=
  \begin{cases}
    1, & z>0,\\
    \alpha, & z<0.
  \end{cases}
\end{equation}

\section{Softmax}

Softmax acts on a whole vector $z\in\mathbb{R}^K$, not on one number:
\begin{equation}
  s_i(z)=\frac{e^{z_i}}{\sum_{j=1}^{K}e^{z_j}}, \qquad i=1,\dots,K .
\end{equation}
The output is a probability vector:
\begin{equation}
  s_i>0, \qquad \sum_{i=1}^K s_i=1 .
\end{equation}
Adding the same constant to every entry changes nothing:
\begin{equation}
  s(z+c\mathbf{1})=s(z),
\end{equation}
so in practice one computes $s(z-\max_j z_j)$ to avoid overflow in $e^{z_i}$.

The Jacobian couples all entries:
\begin{equation}
  \frac{\partial s_i}{\partial z_j}=s_i(\delta_{ij}-s_j),
  \qquad
  J=\operatorname{diag}(s)-s\,s^\top .
\end{equation}
With $K=2$ softmax is the sigmoid of the difference:
\begin{equation}
  s_1=\frac{e^{z_1}}{e^{z_1}+e^{z_2}}=\frac{1}{1+e^{-(z_1-z_2)}}=\sigma(z_1-z_2).
  \label{eq:softmax-sigmoid}
\end{equation}
With the cross-entropy loss $\ell=-\sum_i y_i\log s_i$ and a one-hot label $y$, the gradient is simply
\begin{equation}
  \frac{\partial\ell}{\partial z_j}
  =-\sum_i y_i(\delta_{ij}-s_j)
  =s_j-y_j ,
  \qquad
  \nabla_z\ell=s-y .
\end{equation}
A temperature $T>0$ controls how sharp the output is:
\begin{equation}
  s(z/T)\ \longrightarrow\
  \begin{cases}
    \text{one-hot at } \arg\max_i z_i, & T\to0,\\
    \left(\tfrac1K,\dots,\tfrac1K\right), & T\to\infty .
  \end{cases}
\end{equation}
Because the units are coupled, softmax is used at the output layer for classification,
not as a hidden-layer activation.
